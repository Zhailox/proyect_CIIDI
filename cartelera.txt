// FunciÃ³n para mapear el Trayecto a un data-nivel compatible con Investigaciones.css
// Agregaremos inline styles para t1 y t2 ya que Investigaciones.css solo tiene t3, t4 y maestria
function getAttrNivel($trayecto) {
    $trayectoLower = strtolower(trim($trayecto));
    if (strpos($trayectoLower, 'trayecto i') === 0 && strpos($trayectoLower, 'ii') === false && strpos($trayectoLower, 'iv') === false) return 't1';
    if (strpos($trayectoLower, 'trayecto ii') === 0 && strpos($trayectoLower, 'iii') === false) return 't2';
    if (strpos($trayectoLower, 'trayecto iii') !== false) return 't3';
    if (strpos($trayectoLower, 'trayecto iv') !== false) return 't4';
    return 'maestria';
}

function filtrarLinea(linea, btn) {
    // Limpiar botones

    botones.forEach(b => b.classList.remove('active'));
    // Activar el clickeado
    btn.classList.add('active');

    // Filtrar tarjetas
    const tarjetas = document.querySelectorAll('.post-card');
    tarjetas.forEach(tarjeta => {
        if (linea === 'Todas' || tarjeta.getAttribute('data-linea') === linea) {
            tarjeta.style.display = 'block';
        } else {
            tarjeta.style.display = 'none';
        }
    });
}
